<script>
document.addEventListener('DOMContentLoaded', function() {
    const tabEl = document.getElementById('tournamentTab');
    if (!tabEl) return;

    let groupsData = {};
    let groupMatches = {};

    try {
        groupsData = JSON.parse(document.getElementById('groups-data').textContent);
        groupMatches = JSON.parse(document.getElementById('group-matches-data').textContent);
    } catch (e) {
        console.error("Data parsing error:", e);
    }

    const countryFlags = {
        "Sweden": "se", "England": "gb-eng", "Italy": "it", "Malta": "mt",
        "Wales": "gb-wls", "Scotland": "gb-sct", "Germany": "de", "Hungary": "hu",
        "Republic of Ireland": "ie", "Denmark": "dk", "Serbia": "rs", "Spain": "es",
        "France": "fr", "Portugal": "pt", "Netherlands": "nl", "Belgium": "be",
        "Croatia": "hr", "Switzerland": "ch", "Austria": "at", "Poland": "pl"
    };

    function getFlagCode(teamName) {
        return countryFlags[teamName] || null;
    }

    // --- BULLETPROOF TRANSLATION ENGINE ---
    function resolveTeamName(placeholder, groupId) {
        placeholder = String(placeholder).trim();
        const teams = groupsData[groupId];
        
        if (Array.isArray(teams)) {
            if (/^[A-Za-z]?\d$/.test(placeholder)) {
                const digit = parseInt(placeholder.match(/\d/)[0], 10);
                const index = digit - 1; // A1 becomes index 0
                
                if (index >= 0 && index < teams.length) {
                    return teams[index]; 
                }
            }
        }
        return placeholder; 
    }

    // 1. Hydrate Match List HTML with real names and flags
    document.querySelectorAll('.team-name-display').forEach(el => {
        const groupId = el.getAttribute('data-group');
        const original = el.getAttribute('data-original');
        el.textContent = resolveTeamName(original, groupId);
    });

    document.querySelectorAll('.flag-img').forEach(img => {
        const groupId = img.getAttribute('data-group');
        const original = img.getAttribute('data-original');
        const resolvedName = resolveTeamName(original, groupId);
        
        const code = getFlagCode(resolvedName);
        if (code) {
            img.src = `https://flagcdn.com/w40/${code}.png`;
            img.style.display = 'inline-block';
        }
    });

    // 2. Pre-process match logic for standings
    Object.keys(groupMatches).forEach(groupId => {
        if(Array.isArray(groupMatches[groupId])) {
            groupMatches[groupId].forEach(m => {
                m.home = resolveTeamName(m.home, groupId);
                m.away = resolveTeamName(m.away, groupId);
            });
        }
    });

    // 3. Standings Engine
    function recalculateStandings(groupId) {
        const teams = groupsData[groupId];
        if (!Array.isArray(teams) || teams.length === 0) return;

        let stats = {};
        teams.forEach(team => {
            stats[team] = { played: 0, won: 0, drawn: 0, lost: 0, gf: 0, ga: 0, gd: 0, points: 0 };
        });

        const matches = groupMatches[groupId] || [];
        matches.forEach(m => {
            const homeInput = document.querySelector(`input[name="home_${m.id}"]`);
            const awayInput = document.querySelector(`input[name="away_${m.id}"]`);

            if (homeInput && awayInput && homeInput.value !== '' && awayInput.value !== '') {
                const hg = parseInt(homeInput.value);
                const ag = parseInt(awayInput.value);
                const ht = m.home; 
                const at = m.away;

                if (stats[ht] && stats[at]) {
                    stats[ht].played += 1;
                    stats[at].played += 1;
                    stats[ht].gf += hg;
                    stats[ht].ga += ag;
                    stats[at].gf += ag;
                    stats[at].ga += hg;

                    if (hg > ag) {
                        stats[ht].won += 1;
                        stats[ht].points += 3;
                        stats[at].lost += 1;
                    } else if (hg < ag) {
                        stats[at].won += 1;
                        stats[at].points += 3;
                        stats[ht].lost += 1;
                    } else {
                        stats[ht].drawn += 1;
                        stats[ht].points += 1;
                        stats[at].drawn += 1;
                        stats[at].points += 1;
                    }
                }
            }
        });

        Object.keys(stats).forEach(team => {
            stats[team].gd = stats[team].gf - stats[team].ga;
        });

        const sortedTeams = [...teams].sort((a, b) => {
            if (stats[b].points !== stats[a].points) return stats[b].points - stats[a].points;
            if (stats[b].gd !== stats[a].gd) return stats[b].gd - stats[a].gd;
            return stats[b].gf - stats[a].gf;
        });

        const tableBody = document.querySelector(`#standings-table-g${groupId} tbody`);
        if (!tableBody) return;
        tableBody.innerHTML = '';

        sortedTeams.forEach(team => {
            const s = stats[team];
            const flagCode = getFlagCode(team);
            const flagHtml = flagCode 
                ? `<img src="https://flagcdn.com/w40/${flagCode}.png" width="22" class="border border-1 shadow-sm me-2" style="border-radius: 3px; object-fit: cover;">` 
                : '';

            const row = `
                <tr class="border-bottom">
                    <td class="fw-bold text-dark px-3 py-2 d-flex align-items-center">${flagHtml}${team}</td>
                    <td class="text-center py-2 text-muted">${s.played}</td>
                    <td class="text-center py-2">${s.won}</td>
                    <td class="text-center py-2">${s.drawn}</td>
                    <td class="text-center py-2">${s.lost}</td>
                    <td class="text-center py-2 fw-semibold ${s.gd > 0 ? 'text-success' : s.gd < 0 ? 'text-danger' : ''}">${s.gd > 0 ? '+'+s.gd : s.gd}</td>
                    <td class="text-center py-2 fw-bold text-success fs-6">${s.points}</td>
                </tr>
            `;
            tableBody.insertAdjacentHTML('beforeend', row);
        });
    }

    // 4. Attach Event Listeners for Live Updates
    document.querySelectorAll('.match-input').forEach(input => {
        input.addEventListener('input', function() {
            const groupId = this.getAttribute('data-group');
            recalculateStandings(groupId);
        });
    });

    // 5. Initial calculation on load
    Object.keys(groupsData).forEach(groupId => {
        recalculateStandings(groupId);
    });
});
</script>

{% endblock %}
